const moveBar = () =>{
    const current = document.getElementById("current");
    let width = 0.00;

    const rate = () => {
        width ++;
        current.style.width = `${width}%`;
    };

    setInterval(rate,30);
};


progressBar.addEventListener("click",moveBar);
